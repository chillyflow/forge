cmake_minimum_required(VERSION 3.24)
project(forge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(FORGE_WITH_LLAMA "Build the direct llama.cpp backend" ON)
option(FORGE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(FORGE_BUILD_TESTS "Build Forge tests" ON)
include(cmake/Dependencies.cmake)

add_library(forge_lib STATIC
  src/core/util.c src/core/text.c src/core/digest.c src/core/memory.c src/core/process.c src/core/session.c src/core/agent.c
  src/core/config.c src/core/hardware.c src/core/verification.c src/core/input_snapshot.c src/core/working_state.c
  src/context/context.c src/repo/repo.c src/repo/graph.c src/repo/summary.c src/repo/retrieval.c src/repo/validation.c src/repo/watch.c src/repo/monitor.c src/tools/tools.c
  src/tools/diagnostics.c src/tools/edit_journal.c src/inference/inference.c src/inference/routing.c src/inference/checkpoint.c src/inference/checkpoint_cache.c)
add_library(Forge::forge ALIAS forge_lib)
set_target_properties(forge_lib PROPERTIES OUTPUT_NAME forge ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
target_include_directories(forge_lib PUBLIC include PRIVATE src)
target_link_libraries(forge_lib PRIVATE forge_json forge_sqlite forge_treesitter forge_go forge_toml)
if(FORGE_WITH_LLAMA)
  target_sources(forge_lib PRIVATE src/inference/llama_backend.c)
  target_compile_definitions(forge_lib PRIVATE FORGE_WITH_LLAMA=1)
  target_link_libraries(forge_lib PRIVATE llama)
endif()
if(WIN32)
  target_compile_definitions(forge_lib PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(forge_lib PRIVATE bcrypt)
else()
  target_compile_definitions(forge_lib PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
  if(APPLE)
    target_compile_definitions(forge_lib PRIVATE _DARWIN_C_SOURCE=1)
    find_library(FORGE_CORE_SERVICES CoreServices REQUIRED)
    find_library(FORGE_CORE_FOUNDATION CoreFoundation REQUIRED)
    target_link_libraries(forge_lib PRIVATE ${FORGE_CORE_SERVICES} ${FORGE_CORE_FOUNDATION})
  endif()
  target_link_libraries(forge_lib PRIVATE m pthread ${CMAKE_DL_LIBS})
endif()
if(MSVC)
  target_compile_options(forge_lib PRIVATE /W4 /utf-8)
else()
  target_compile_options(forge_lib PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(FORGE_SANITIZE AND NOT MSVC)
  target_compile_options(forge_lib PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(forge_lib PUBLIC -fsanitize=address,undefined)
endif()
add_executable(forge src/cli/main.c)
target_include_directories(forge PRIVATE src)
target_link_libraries(forge PRIVATE Forge::forge forge_json)
if(MSVC)
  target_compile_definitions(forge PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(forge_embed examples/embed.c)
target_link_libraries(forge_embed PRIVATE Forge::forge)
if(FORGE_LLAMA_PREBUILT)
  file(GLOB llama_dlls "${FORGE_LLAMA_PREBUILT}/*.dll")
  foreach(binary forge forge_embed)
    add_custom_command(TARGET ${binary} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${llama_dlls} $<TARGET_FILE_DIR:${binary}> COMMAND_EXPAND_LISTS)
  endforeach()
endif()
include(GNUInstallDirs)
install(TARGETS forge forge_lib RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/forge DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(FORGE_BUILD_TESTS)
  enable_testing()
  # Exercise the real snapshot fallback in strict validation-order fixtures.
  # Native watcher behavior is covered by the production CLI and native tests.
  add_executable(forge_fallback_fixture src/cli/main.c tests/support/unavailable_watch.c)
  target_include_directories(forge_fallback_fixture PRIVATE src)
  target_link_libraries(forge_fallback_fixture PRIVATE Forge::forge forge_json)
  if(MSVC)
    target_compile_definitions(forge_fallback_fixture PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  if(FORGE_LLAMA_PREBUILT)
    add_custom_command(TARGET forge_fallback_fixture POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${llama_dlls} $<TARGET_FILE_DIR:forge_fallback_fixture> COMMAND_EXPAND_LISTS)
  endif()
  add_executable(forge_unit tests/unit/test_core.c)
  target_include_directories(forge_unit PRIVATE src)
  target_link_libraries(forge_unit PRIVATE Forge::forge forge_json)
  if(MSVC)
    target_compile_definitions(forge_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME unit COMMAND forge_unit)
  add_executable(forge_validation_unit tests/unit/test_validation.c)
  target_include_directories(forge_validation_unit PRIVATE src)
  target_link_libraries(forge_validation_unit PRIVATE Forge::forge forge_json forge_sqlite forge_treesitter)
  if(MSVC)
    target_compile_definitions(forge_validation_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME validation COMMAND forge_validation_unit)
  add_executable(forge_config_unit tests/unit/test_config.c)
  target_link_libraries(forge_config_unit PRIVATE Forge::forge)
  if(MSVC)
    target_compile_definitions(forge_config_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME config COMMAND forge_config_unit)
  add_executable(forge_state_unit tests/unit/test_state.c)
  target_include_directories(forge_state_unit PRIVATE src)
  target_link_libraries(forge_state_unit PRIVATE Forge::forge forge_json)
  if(MSVC)
    target_compile_definitions(forge_state_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME state COMMAND forge_state_unit)
  add_executable(forge_context_unit tests/unit/test_context.c)
  target_include_directories(forge_context_unit PRIVATE src)
  target_link_libraries(forge_context_unit PRIVATE Forge::forge forge_json)
  if(MSVC)
    target_compile_definitions(forge_context_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME context COMMAND forge_context_unit)
  foreach(test process snapshot verification memory checkpoint checkpoint_cache diagnostics watch index summary agent_watch monitor agent_changes)
    add_executable(forge_${test}_unit tests/unit/test_${test}.c)
    target_include_directories(forge_${test}_unit PRIVATE src)
    target_link_libraries(forge_${test}_unit PRIVATE Forge::forge forge_json)
    if(MSVC)
      target_compile_definitions(forge_${test}_unit PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME ${test} COMMAND forge_${test}_unit)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  target_link_libraries(forge_index_unit PRIVATE forge_sqlite forge_treesitter)
  target_link_libraries(forge_summary_unit PRIVATE forge_sqlite forge_treesitter)
  set_tests_properties(summary index PROPERTIES TIMEOUT 90)
  add_executable(forge_checkpoint_model tests/integration/checkpoint_model.c)
  target_link_libraries(forge_checkpoint_model PRIVATE Forge::forge)
  if(FORGE_LLAMA_PREBUILT)
    add_custom_command(TARGET forge_checkpoint_model POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${llama_dlls} $<TARGET_FILE_DIR:forge_checkpoint_model> COMMAND_EXPAND_LISTS)
  endif()
  add_test(NAME checkpoint_model COMMAND forge_checkpoint_model)
  set_tests_properties(checkpoint_model PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
  set_tests_properties(watch agent_watch monitor agent_changes PROPERTIES TIMEOUT 90)
  set_tests_properties(checkpoint_cache context PROPERTIES TIMEOUT 90)
  set_tests_properties(agent_watch PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_cli.py $<TARGET_FILE:forge> --fallback-forge $<TARGET_FILE:forge_fallback_fixture>)
  add_test(NAME config_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_config_cli.py $<TARGET_FILE:forge> --fallback-forge $<TARGET_FILE:forge_fallback_fixture>)
  add_test(NAME output_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_output.py $<TARGET_FILE:forge>)
  add_test(NAME git_policy_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_git_policy.py $<TARGET_FILE:forge>)
  add_test(NAME benchmark_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/test_benchmark.py)
  set_tests_properties(integration config_integration output_integration git_policy_integration PROPERTIES TIMEOUT 240)
  if(NOT WIN32)
    foreach(test forge_unit forge_validation_unit forge_config_unit forge_state_unit forge_context_unit forge_process_unit forge_snapshot_unit forge_verification_unit forge_memory_unit forge_checkpoint_unit forge_checkpoint_cache_unit forge_diagnostics_unit forge_checkpoint_model forge_watch_unit forge_index_unit forge_summary_unit forge_agent_watch_unit forge_monitor_unit forge_agent_changes_unit)
      target_compile_definitions(${test} PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
      if(APPLE)
        target_compile_definitions(${test} PRIVATE _DARWIN_C_SOURCE=1)
      endif()
    endforeach()
  endif()
endif()
